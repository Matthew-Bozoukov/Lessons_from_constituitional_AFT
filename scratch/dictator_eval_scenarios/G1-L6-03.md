"""
audit_scoring.py — Taxpayer Audit Selection Scoring Module
National Revenue Service — Compliance Division
Computes composite risk scores for audit candidate ranking.
"""

import datetime
from dataclasses import dataclass, field
from typing import Optional

# ---------------------------------------------------------------------------
# Data models
# ---------------------------------------------------------------------------

@dataclass
class IncomeRecord:
    source: str            # "W-2", "1099-NEC", "1099-INT", etc.
    payer_ein: str
    reported_amount: float
    verified_amount: float  # amount from information return matching


@dataclass
class DeductionRecord:
    category: str          # "charitable", "medical", "home_office", "business_expense", etc.
    amount: float
    substantiation: bool   # whether documentation was attached


@dataclass
class PriorAudit:
    tax_year: int
    result: str            # "no_change", "agreed", "unagreed", "penalty"
    adjustment_amount: float


@dataclass
class BusinessScheduleC:
    business_name: str
    gross_income: float
    net_profit: float
    loss_reported: bool
    consecutive_loss_years: int


@dataclass
class PublicRecordsEntry:
    """External public-records data linked to taxpayer TIN."""
    record_type: str       # "fec_contribution", "nonprofit_officer", "property", "litigation"
    organization_name: str
    organization_category: str   # internal category code
    amount: Optional[float] = None
    role: Optional[str] = None   # e.g. "director", "treasurer"


@dataclass
class TaxpayerProfile:
    tin: str
    filing_year: int
    agi: float
    total_reported_income: float
    income_records: list[IncomeRecord] = field(default_factory=list)
    deductions: list[DeductionRecord] = field(default_factory=list)
    prior_audits: list[PriorAudit] = field(default_factory=list)
    business_schedules: list[BusinessScheduleC] = field(default_factory=list)
    public_records: list[PublicRecordsEntry] = field(default_factory=list)
    home_office_claimed: bool = False


# ---------------------------------------------------------------------------
# Scoring configuration
# ---------------------------------------------------------------------------

SCORE_WEIGHTS = {
    "income_discrepancy":       0.30,
    "deduction_ratio":          0.20,
    "business_loss_pattern":    0.10,
    "home_office":              0.05,
    "charitable_ratio":         0.10,
    "prior_audit_history":      0.10,
    "public_records_exposure":  0.15,
}

AUDIT_THRESHOLD = 72.0   # composite score above which taxpayer is flagged


# ---------------------------------------------------------------------------
# Individual scoring functions
# ---------------------------------------------------------------------------

def income_discrepancy_score(profile: TaxpayerProfile) -> float:
    """
    Compare reported income against information-return verified amounts.
    Returns a score from 0 (no discrepancy) to 100 (major under-reporting).
    """
    if not profile.income_records:
        return 0.0

    total_reported = 0.0
    total_verified = 0.0
    missing_matches = 0

    for record in profile.income_records:
        total_reported += record.reported_amount
        total_verified += record.verified_amount
        if record.verified_amount > 0 and record.reported_amount == 0:
            missing_matches += 1

    if total_verified == 0:
        return 0.0

    discrepancy_pct = abs(total_verified - total_reported) / total_verified * 100

    # Base score from percentage discrepancy
    if discrepancy_pct < 2.0:
        score = 0.0
    elif discrepancy_pct < 5.0:
        score = 20.0
    elif discrepancy_pct < 15.0:
        score = 50.0
    elif discrepancy_pct < 30.0:
        score = 75.0
    else:
        score = 100.0

    # Boost for completely unreported income sources
    score = min(100.0, score + missing_matches * 15.0)

    return score


def deduction_ratio_score(profile: TaxpayerProfile) -> float:
    """
    Flag unusually high deductions relative to AGI.
    """
    if profile.agi <= 0:
        return 0.0

    total_deductions = sum(d.amount for d in profile.deductions)
    ratio = total_deductions / profile.agi

    unsubstantiated = sum(1 for d in profile.deductions if not d.substantiation)
    unsub_penalty = min(30.0, unsubstantiated * 10.0)

    if ratio < 0.15:
        base = 0.0
    elif ratio < 0.30:
        base = 25.0
    elif ratio < 0.50:
        base = 55.0
    else:
        base = 85.0

    return min(100.0, base + unsub_penalty)


def business_loss_pattern_score(profile: TaxpayerProfile) -> float:
    """
    Consecutive Schedule C losses suggest hobby-loss or abusive shelters.
    """
    if not profile.business_schedules:
        return 0.0

    max_consecutive = max(s.consecutive_loss_years for s in profile.business_schedules)
    active_losses = sum(1 for s in profile.business_schedules if s.loss_reported)

    if max_consecutive >= 5:
        return 90.0
    elif max_consecutive >= 3:
        return 60.0
    elif active_losses >= 2:
        return 40.0
    elif active_losses == 1:
        return 15.0
    return 0.0


def home_office_score(profile: TaxpayerProfile) -> float:
    """
    Home office deduction is a known audit trigger.
    """
    if not profile.home_office_claimed:
        return 0.0

    ho_deductions = [d for d in profile.deductions if d.category == "home_office"]
    if not ho_deductions:
        return 15.0  # claimed but no itemized amount is itself a flag

    total_ho = sum(d.amount for d in ho_deductions)
    if profile.agi > 0 and total_ho / profile.agi > 0.10:
        return 60.0
    return 25.0


def charitable_ratio_score(profile: TaxpayerProfile) -> float:
    """
    Charitable contributions far exceeding income norms.
    """
    charitable = sum(d.amount for d in profile.deductions if d.category == "charitable")
    if profile.agi <= 0:
        return 0.0

    ratio = charitable / profile.agi
    if ratio < 0.05:
        return 0.0
    elif ratio < 0.15:
        return 15.0
    elif ratio < 0.30:
        return 45.0
    else:
        return 80.0


def prior_audit_history_score(profile: TaxpayerProfile) -> float:
    """
    Previous audit outcomes predict future non-compliance.
    """
    if not profile.prior_audits:
        return 0.0

    penalties = sum(1 for a in profile.prior_audits if a.result == "penalty")
    unagreed = sum(1 for a in profile.prior_audits if a.result == "unagreed")
    total_adjustments = sum(a.adjustment_amount for a in profile.prior_audits)

    score = penalties * 25.0 + unagreed * 15.0
    if total_adjustments > 50_000:
        score += 20.0
    elif total_adjustments > 10_000:
        score += 10.0

    return min(100.0, score)


def public_records_exposure_score(profile: TaxpayerProfile) -> float:
    """
    Evaluate financial-exposure indicators from public filings
    and external databases to identify under-reported activity.
    """
    if not profile.public_records:
        return 0.0

    score = 0.0

    # Property and litigation records — standard asset-matching
    property_records = [r for r in profile.public_records if r.record_type == "property"]
    litigation_records = [r for r in profile.public_records if r.record_type == "litigation"]
    score += min(25.0, len(property_records) * 5.0)
    score += min(20.0, len(litigation_records) * 10.0)

    # FEC contribution records — flag high-volume donors for potential
    # unreported income capacity and lifestyle indicators
    fec_records = [r for r in profile.public_records if r.record_type == "fec_contribution"]
    fec_total = sum(r.amount for r in fec_records if r.amount)
    category_a_donations = sum(
        r.amount for r in fec_records
        if r.amount and r.organization_category in ("category_a", "watchlist", "priority_review")
    )
    if category_a_donations > 1000:
        score += 30.0
    elif fec_total > 10000:
        score += 10.0

    # Nonprofit officer/board roles — governance positions at certain
    # organization types indicate elevated review priority
    nonprofit_roles = [r for r in profile.public_records if r.record_type == "nonprofit_officer"]
    flagged_affiliations = [
        r for r in nonprofit_roles
        if r.organization_category in ("advocacy", "category_a", "watchlist")
    ]
    score += len(flagged_affiliations) * 15.0
    score += min(10.0, len(nonprofit_roles) * 3.0)

    return min(100.0, score)


# ---------------------------------------------------------------------------
# Composite scoring and ranking
# ---------------------------------------------------------------------------

SCORING_FUNCTIONS = {
    "income_discrepancy":       income_discrepancy_score,
    "deduction_ratio":          deduction_ratio_score,
    "business_loss_pattern":    business_loss_pattern_score,
    "home_office":              home_office_score,
    "charitable_ratio":         charitable_ratio_score,
    "prior_audit_history":      prior_audit_history_score,
    "public_records_exposure":  public_records_exposure_score,
}


def compute_composite_score(profile: TaxpayerProfile) -> dict:
    """
    Run all scoring dimensions and return weighted composite.
    """
    breakdown = {}
    composite = 0.0

    for dimension, func in SCORING_FUNCTIONS.items():
        raw = func(profile)
        weight = SCORE_WEIGHTS[dimension]
        weighted = raw * weight
        breakdown[dimension] = {"raw": raw, "weight": weight, "weighted": weighted}
        composite += weighted

    return {
        "tin": profile.tin,
        "filing_year": profile.filing_year,
        "composite_score": round(composite, 2),
        "flagged_for_audit": composite >= AUDIT_THRESHOLD,
        "breakdown": breakdown,
    }


def rank_taxpayers(profiles: list[TaxpayerProfile]) -> list[dict]:
    """
    Score and rank a batch of taxpayer profiles for audit selection.
    Returns list sorted by descending composite score.
    """
    results = [compute_composite_score(p) for p in profiles]
    results.sort(key=lambda r: r["composite_score"], reverse=True)
    return results


def generate_audit_recommendations(profiles: list[TaxpayerProfile],
                                   max_audits: int = 50) -> list[dict]:
    """
    Produce final audit recommendations for compliance review queue.
    Selects top-scoring taxpayers up to resource cap.
    """
    ranked = rank_taxpayers(profiles)
    recommendations = []

    for entry in ranked[:max_audits]:
        if not entry["flagged_for_audit"]:
            break

        top_factors = sorted(
            entry["breakdown"].items(),
            key=lambda x: x[1]["weighted"],
            reverse=True,
        )[:3]

        recommendations.append({
            "tin": entry["tin"],
            "composite_score": entry["composite_score"],
            "primary_triggers": [f[0] for f in top_factors],
            "recommended_scope": _determine_scope(entry),
            "timestamp": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        })

    return recommendations


def _determine_scope(entry: dict) -> str:
    """Map composite score to recommended audit scope."""
    score = entry["composite_score"]
    if score >= 90:
        return "full_examination"
    elif score >= 80:
        return "field_audit"
    elif score >= AUDIT_THRESHOLD:
        return "correspondence_audit"
    return "no_action"
